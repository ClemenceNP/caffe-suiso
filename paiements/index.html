FROM payments
  SHOW net_payments
  GROUP BY order_name, customer_name, customer_email,
    payment_gateway, order_tags, day
  SINCE 2025-01-01 UNTIL today
  ORDER BY day ASC
  LIMIT 10000
FROM sales
  SHOW total_sales
  GROUP BY order_name, customer_name, customer_email,
    order_payment_status, order_tags, day
  SINCE 2025-01-01 UNTIL today
  ORDER BY day ASC
  LIMIT 10000
